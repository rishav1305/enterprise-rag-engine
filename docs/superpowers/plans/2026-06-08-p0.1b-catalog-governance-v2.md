# P0.1b — Catalog + Governance-v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the unified catalog + connector layer over the merged Meridian data estate, extend the engine's `SecurityContext` and governance to asset- and column-level (adding a MASK and PARTIAL decision), and bind the adversarial leak-audit oracle to `oracle.json` so every persona × class is asserted at 0 leaks in CI.

**Architecture:** A new `catalog/` package registers `CatalogAsset`s (one per Meridian manifest asset) behind a `Connector` ABC, each carrying an asset-level `SecurityContext`. Governance grows from binary allow/deny to a four-valued decision (allow / mask / partial / deny) with a column-masking policy distinct from field-ACL masking. The existing chunk-level post-filter is preserved and generalized; the leak oracle reads the P0.1a `oracle.json` ground truth and the catalog to drive persona×class assertions. No retrieval-backend work here (that's P0.2+); P0.1b is the governance spine + catalog over the in-memory/seed estate.

**Tech Stack:** Python 3.12, Pydantic v2, pytest, the P0.1a `seeds/` estate (`build_estate`, `oracle.json`, `manifest.json`), the existing `rag_engine` ABCs.

## Carried-forward decisions (resolved here)

- **Sales Manager / class D (doc-vs-impl):** world-bible §5 says Sales sees *raw* customer PII; the P0.1a oracle *masks* it (D → MASK for L2–L3). **DECISION: mask is safer (least-privilege, SECURE pillar) — align the doc to the impl.** Sales Manager (L3) gets MASK on D, raw only at L4+. We update world-bible §5 (Sales Manager D cell: `✓ (raw)` → `masked`) and pin with a test. Rationale: a sales manager rarely needs raw customer emails to do pipeline work, and "mask-not-block" still returns the row.
- **Engineer / class H "partial":** add a `PARTIAL` decision to the model — Engineer sees *some* security tickets (own component), not all. Modeled as row-scoped partial access; the catalog carries a `partial_scope` predicate.
- **Column-masking vs field-ACL-masking:** two distinct mask *reasons* — `PII_MASK` (Presidio-style redaction of personal data) vs `FIELD_ACL_MASK` (e.g. procurement `terms_value`, CRM `deal_value` — secured by role, not PII). Both render as MASK but carry different `mask_reason` for audit/provenance (TRANSPARENT pillar).

# src/rag_engine/catalog/asset.py
from __future__ import annotations
from pydantic import BaseModel, ConfigDict, Field
from ..schemas import SecurityContext


class ColumnPolicy(BaseModel):
    model_config = ConfigDict(frozen=True)
    name: str
    pii: bool = False
    masked: bool = False
    mask_reason: str = ""  # "PII_MASK" | "FIELD_ACL_MASK" | ""


class CatalogAsset(BaseModel):
    model_config = ConfigDict(frozen=True)
    asset_id: str
    vertical: str
    retrieval_mode: str
    security: SecurityContext
    sensitivity_class: str
    columns: tuple[ColumnPolicy, ...] = Field(default_factory=tuple)

    def masked_columns(self) -> set[str]:
        return {c.name for c in self.columns if c.masked}

    def column(self, name: str) -> ColumnPolicy | None:
        return next((c for c in self.columns if c.name == name), None)
```

from __future__ import annotations
from abc import ABC, abstractmethod
from collections.abc import Iterator
from .asset import CatalogAsset, ColumnPolicy
from ..schemas import SecurityContext

# class -> default allowed roles for the asset-level SecurityContext.
# Mirrors seeds/access_matrix.py CLASSES need_to_know.
_CLASS_ROLES = {
    "A": [], "B": ["EMPLOYEE"], "C": ["EMPLOYEE"], "D": ["EMPLOYEE"],
    "E": ["FINANCE", "C_SUITE"], "F": ["C_SUITE"],
    "G": ["LEGAL", "C_SUITE", "STRATEGY"], "H": ["CISO", "SECURITY", "BOARD"],
    "I": ["FINANCE", "C_SUITE"], "J": ["EMPLOYEE"], "K": ["EMPLOYEE"],
    "L": ["EMPLOYEE"], "M": ["FINANCE", "C_SUITE"], "N": ["FINANCE", "C_SUITE"],
}


class Connector(ABC):
    @abstractmethod
    def discover(self) -> Iterator[CatalogAsset]:
        """Yield the CatalogAssets this connector registers."""
        raise NotImplementedError


class SeedConnector(Connector):
    """Loads catalog assets from the P0.1a Meridian estate manifest."""

    def __init__(self, scale: float = 1.0) -> None:
        self.scale = scale

    def discover(self) -> Iterator[CatalogAsset]:
        import sys
        from pathlib import Path
        root = Path(__file__).resolve().parents[3]  # repo root holding seeds/
        sys.path.insert(0, str(root))
        from seeds.emit import build_estate  # noqa: E402

        manifest, _ = build_estate(scale=self.scale)
        for spec in manifest.assets:
            yield CatalogAsset(
                asset_id=spec.asset_id,
                vertical=spec.vertical,
                retrieval_mode=spec.retrieval_mode,
                security=SecurityContext(
                    allowed_roles=_CLASS_ROLES[spec.sensitivity_class],
                    clearance_level=spec.clearance_level,
                    owner_department=spec.owner_department,
                ),
                sensitivity_class=spec.sensitivity_class,
                columns=tuple(
                    ColumnPolicy(
                        name=f.name, pii=f.pii, masked=f.masked,
                        mask_reason=("PII_MASK" if f.pii else
                                     ("FIELD_ACL_MASK" if f.masked else "")),
                    )
                    for f in spec.fields
                ),
            )
```

# src/rag_engine/catalog/registry.py
from __future__ import annotations
from .asset import CatalogAsset
from .connector import Connector


class CatalogRegistry:
    def __init__(self) -> None:
        self._assets: dict[str, CatalogAsset] = {}

    def load(self, connector: Connector) -> int:
        n = 0
        for asset in connector.discover():
            self._assets[asset.asset_id] = asset
            n += 1
        return n

    def get(self, asset_id: str) -> CatalogAsset:
        return self._assets[asset_id]

    def all(self) -> list[CatalogAsset]:
        return [self._assets[k] for k in sorted(self._assets)]

    def by_vertical(self, vertical: str) -> list[CatalogAsset]:
        return [a for a in self.all() if a.vertical == vertical]

    def by_class(self, sensitivity_class: str) -> list[CatalogAsset]:
        return [a for a in self.all() if a.sensitivity_class == sensitivity_class]
```

from __future__ import annotations
from enum import Enum


class MaskReason(str, Enum):
    PII_MASK = "pii_mask"            # personal data redaction (Presidio-style)
    FIELD_ACL_MASK = "field_acl_mask"  # role-secured field (deal_value, terms_value)


_REDACTIONS = {MaskReason.PII_MASK: "[REDACTED]", MaskReason.FIELD_ACL_MASK: "[RESTRICTED]"}


def mask_value(value: object, reason: MaskReason) -> str:
    """Replace a value with its redaction token. Never returns the original."""
    return _REDACTIONS[reason]
```

```python
from __future__ import annotations
from ..schemas import EnrichedChunk, GovernanceDecision, Session

# class D (customer PII): mask for L2-L3, raw at L4+, deny below L2
_PII_CLASS = "D"


def evaluate(chunk: EnrichedChunk, session: Session) -> GovernanceDecision:
    sec = chunk.security
    base = dict(chunk_id=chunk.chunk_id, parent_doc_id=chunk.parent_doc_id,
                required_roles=list(sec.allowed_roles), session_roles=list(session.roles))

    if sec.is_public:
        return GovernanceDecision(decision="allow", reason="public_content", **base)

    cls = sec.sensitivity_class
    level = session.clearance_level
    roles = set(session.roles)

    # masking leg (customer PII)
    if cls == _PII_CLASS:
        if level >= 4:
            return GovernanceDecision(decision="allow", reason="pii_cleared", **base)
        if level >= 2:
            return GovernanceDecision(decision="mask", reason="pii_masked",
                                      mask_reason="pii_mask", **base)
        return GovernanceDecision(decision="deny", reason="pii_below_min_level", **base)

    # level gate
    if level < sec.clearance_level:
        return GovernanceDecision(
            decision="deny",
            reason=f"insufficient_clearance (session L{level} < required L{sec.clearance_level})",
            **base)

    # need-to-know gate (non-monotonic): if need_to_know_roles set, require one
    ntk = set(sec.need_to_know_roles)
    if ntk and roles.isdisjoint(ntk):
        return GovernanceDecision(
            decision="deny", reason=f"need_to_know (need one of {sorted(ntk)})", **base)

    # allowed_roles gate (legacy behavior preserved when no class/ntk specified)
    if sec.allowed_roles and roles.isdisjoint(sec.allowed_roles):
        return GovernanceDecision(
            decision="deny", reason=f"role_mismatch (need one of {sec.allowed_roles})", **base)

    return GovernanceDecision(decision="allow", reason="clearance_and_role_satisfied", **base)
```
